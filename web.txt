function showTable() {
var num = prompt("Enter a number for multiplication table:");
var table = "";
for (var i = 1; i <= 10; i++) {
table += num + " × " + i + " = " + (num * i) + "<br>";
} //document.write(table);
document.getElementById("result").innerHTML = table;
}

Vehicle Dealer.html
<!DOCTYPE html>
<html>
<head>
<title>XYZ Vehicle Dealer</title>
<script>
function showDateTime() {
var now = new Date();
document.getElementById("datetime").innerHTML = now.toLocaleString();
}
function showVehicleDetails() {
var vehicle = document.getElementById("vehicleSelect").value;
var details = "";
if (vehicle === "car") {
details = "Car - Model: Toyota Corolla, Price: ₹12,00,000, Fuel: Petrol";
} else if (vehicle === "bike") {
details = "Bike - Model: Royal Enfield Classic, Price: ₹1,50,000, Fuel: Petrol";
} else if (vehicle === "suv") {
details = "SUV - Model: Hyundai Creta, Price: ₹15,00,000, Fuel: Diesel";
} else {
details = "Please select a vehicle";
}
document.getElementById("vehicleDetails").innerHTML = details;
}
</script>
</head>
<body onload="showDateTime()">
<h1>XYZ Vehicle Dealer</h1>
<!-- Display Current Date & Time -->
<p>Current Date and Time: <span id="datetime"></span></p>
<!-- Vehicle Selection -->
<h3>Select a Vehicle:</h3>
<select id="vehicleSelect">
<option value="">--Select--</option>
<option value="car">Car</option>
<option value="bike">Bike</option>
<option value="suv">SUV</option>
</select>
<br><br>
<input type="button" value="Show Details" onclick="showVehicleDetails()">
<!-- Display Vehicle Details -->
<h3>Vehicle Details:</h3>
<p id="vehicleDetails"></p>
</body>
</html>
