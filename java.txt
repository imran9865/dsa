import java.io.File;

public class SoundPlayer extends JFrame implements ActionListener {
JButton b1,b2,b3;
Clip clip;

b1=new JButton("Play");
b2=new JButton("Loop");
b3=new JButton("Stop");
add(b1); add(b2); add(b3);
setLayout(new java.awt.FlowLayout());
setSize(300,150);
setVisible(true);
setDefaultCloseOperation(EXIT_ON_CLOSE);
b1.addActionListener(this);
b2.addActionListener(this);
b3.addActionListener(this);
try {
AudioInputStream a=AudioSystem.getAudioInputStream(new File("song.wav"));
clip=AudioSystem.getClip();
clip.open(a);
catch(Exception e) {
System.out.println("Error loading audio file: " + e.getMessage());
} }
public void actionPerformed(ActionEvent e) {
if(e.getSource()==b1) {
clip.setFramePosition(0);
clip.start();
}
if(e.getSource()==b2)
clip.loop(Clip.LOOP_CONTINUOUSLY);
if(e.getSource()==b3)
clip.stop();
}
public static void main(String[] args) {
new SoundPlayer();
}
}

import java.rmi.registry.*;

//Program 9. Program in Java to convert an image in RGB to a Grayscale image.
//Driver.java
import java.awt.*;
import java.awt.image.BufferedImage;
import java.io.*;
import javax.imageio.ImageIO;
public class Driver {
BufferedImage img;
int width;
int height;
public Driver() {
try {
File input = new File("vijay.jpg");
img = ImageIO.read(input);
width = img.getWidth();
height = img.getHeight();
for(int i=0; i<height; i++) {
for (int j=0; j<width; j++) {
Color c = new Color(img.getRGB(j, i));
int r = (int)(c.getRed() * 0.299);
int g = (int)(c.getGreen() * 0.587);
int b = (int)(c.getBlue() * 0.114);
Color newcolor = new Color(r+g+b, r+g+b, r+g+b);
img.setRGB(j, i, newcolor.getRGB());
}
}
File output = new File("GrayVijay.jpg");
ImageIO.write(img, "jpg", output);
}
catch(Exception e) { e.printStackTrace(); }
}
public static void main(String[] args) {
Driver obj = new Driver();
}
}
